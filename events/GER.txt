news_event = {

	id = TODgermany.1003
	hidden = yes
	
	fire_only_once = yes
	
	trigger = {
		# 德国须已完成中欧整合，且尚未进入法国战争；将条件分别否定，
		# 避免“任一条件不成立”便过早触发事件。
		NOT = { country_exists = AUS }
		NOT = { country_exists = BAV }
		GER = { NOT = { has_war_with = FRA } }
		NOT = { has_global_flag = defa_kaizhan }
		country_exists = FRA
	}


	mean_time_to_happen = {
		days = 120
	}
	
	immediate = {
		GER = {news_event = { id = TODgermany.1004 days = 120 }}
	}
	
	option = {
		name = TODgermany.1003.a
	}
}

	desc =TODgermany.1004.d
	picture = GFX_report_event_chinese_soldiers_fighting
	
	major = yes

		hidden_effect = {
			# 波兰若已在匈牙利领导的东墙，不得因西欧事件中途退出。

					}
				}
				POL = { leave_faction = yes }
			}
			set_global_flag = defa_kaizhan
			ENG = {country_event = { id = TODengland.1 days = 30 }}
			FRA = {
				remove_ideas = FRA_paomopo
				remove_ideas = FRA_shilizhe
				remove_ideas = FRA_fubai
			}
		}
	}
	option = { 
		trigger = {NOT = {tag = GER}}
		name = TODgermany.1004.a
	}
	option = { 
		trigger = {tag = GER}
		name = TODgermany.1004.b
		declare_war_on = { target = FRA type = annex_everything }
		declare_war_on = { target = ITA type = annex_everything }
		declare_war_on = { target = SPR type = annex_everything }
	}
}

	}
}
