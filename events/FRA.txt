	id = TODfrance.1
	title = TODfrance.1.t
	desc =TODfrance.1.d
	picture = GFX_report_event_chinese_soldiers_fighting

	fire_only_once = yes

		tag = FRA
		has_war_with = GER

            16 = { is_controlled_by = GER }
            16 = { is_controlled_by = ENG }
        }
		# 仅在接近投降时处理流亡/傀儡，避免战争打到一半就触发白和。
		surrender_progress > 0.8

	}

	option = { # 成为德国傀儡
		name = TODfrance.1.a
		trigger = { is_ai = no }
		set_country_flag = become_deguokuilei
		GER = { country_event = TODfrance.14 }
		custom_effect_tooltip = TODfrance.1_tt
	}
	option = { # 流亡北非
		name = TODfrance.1.b
		ai_chance = {
			factor = 100
		}
		set_global_flag = fall_of_france
		save_global_event_target_as = free_france
		GER = { country_event = TODfrance.14 }
		custom_effect_tooltip = TODfrance.1.b_tt
	}
}

country_event = {
	id = TODfrance.15
	title = TODfrance.15.t
	desc = TODfrance.15.d
	picture = GFX_report_event_german_parade_paris
	fire_only_once = yes
	mean_time_to_happen = { days = 1 }
	trigger = {
		tag = FRA
		NOT = { has_war_with = RUS }
		country_exists = HUN
		country_exists = RUS
		OR = {
			NOT = { country_exists = GER }
			GER = { is_puppet_of = FRA }
		}
		HUN = {
			is_faction_leader = yes
			has_war_with = RUS
		}
		RUS = { NOT = { has_capitulated = yes } }
	}
	option = {
		name = TODfrance.15.a
		add_to_war = {
			targeted_alliance = HUN
			enemy = RUS
			hostility_reason = asked_to_join
		}
	}
}

# 法国在和平会议中取得英德傀儡后，立即落实既定的中欧分区，避免 AI 和谈遗漏附庸国。
country_event = {
	id = TODfrance.16
	title = TODfrance.16.t
	desc = TODfrance.16.d
	picture = GFX_report_event_german_parade_paris
	fire_only_once = yes
	mean_time_to_happen = { days = 1 }
	trigger = {
		tag = FRA
		country_exists = GER
		country_exists = ENG
		GER = { is_puppet_of = FRA }
		ENG = { is_puppet_of = FRA }
		NOT = { has_war_with = GER }
		NOT = { has_war_with = ENG }
	}
	option = {
		name = TODfrance.16.a
		# 德国傀儡持有三国核心，因此先在其领土上释放，再由法国直接确立宗主关系。
		if = { limit = { NOT = { country_exists = RIN } } GER = { release = RIN } }
		if = { limit = { country_exists = RIN } set_autonomy = { target = RIN autonomous_state = autonomy_puppet } }
		if = { limit = { NOT = { country_exists = BAV } } GER = { release = BAV } }
		if = { limit = { country_exists = BAV } set_autonomy = { target = BAV autonomous_state = autonomy_puppet } }
		if = { limit = { NOT = { country_exists = AUS } } GER = { release = AUS } }
		if = { limit = { country_exists = AUS } set_autonomy = { target = AUS autonomous_state = autonomy_puppet } }
		set_global_flag = TOD_FRA_generation_peace_settlement
		news_event = { id = TODfrance.17 days = 1 }
	}
}
