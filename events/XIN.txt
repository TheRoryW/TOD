country_event = {
	id = TOD_XIN.111
	hidden = yes
	fire_only_once = yes
	mean_time_to_happen = { days = 1 }
	trigger = {
		tag = XIN
		NOT = { has_country_flag = XIN_burma_puppet_settlement_completed }
		NOT = { has_war = yes }
		NOT = { country_exists = BRM }
		NOT = {
			any_state = {
				is_core_of = BRM
				NOT = { is_owned_by = ROOT }
			}
		}
	}
	option = {
		name = TOD_XIN.111.a
		release_puppet = BRM
		set_autonomy = {
			target = BRM
			autonomous_state = autonomy_puppet
		}
		set_country_flag = XIN_burma_puppet_settlement_completed
	}
}
