# 1938 年下半年：北欧阶段结束后，帝俄发动第一次冬季战争。
country_event = {
	id = TODRussia.5
	title = TODRussia.5.t
	desc = TODRussia.5.d
	picture = GFX_report_event_chinese_soldiers_fighting
	fire_only_once = yes
	trigger = {
		tag = RUS
		date > 1938.8.31
		OR = {
			has_country_flag = TOD_RUS_caucasus_war_started
			NOT = { country_exists = CAU }
		}
		OR = {
			has_global_flag = TOD_CAU_war_concluded
			NOT = { country_exists = CAU }
		}
		# 不再等待芬兰—瑞典旧北方战争的结算；俄罗斯先完成芬兰阶段。
		# 高加索战争结束后才进入冬季战争；不能只按日期跳过未完成的战事。
		OR = {
			NOT = { country_exists = CAU }
			CAU = {
				NOT = { has_war_with = ARM }
				NOT = { has_war_with = GEO }
				NOT = { has_war_with = AZR }
			}
		}
		# 北方战争尚未结算时先攻芬兰；已结算时只攻击实际持有芬兰核心地的一方。
		OR = {
			AND = {
				NOT = { has_global_flag = TOD_great_northern_war_concluded }
			}
			AND = {
				has_global_flag = TOD_great_northern_war_concluded
				any_state = {
					is_core_of = FIN
					is_owned_by = FIN
				}
			}
			AND = {
				has_global_flag = TOD_great_northern_war_concluded
				any_state = {
					is_core_of = FIN
					is_owned_by = SWE
				}
			}
			AND = {
				has_global_flag = TOD_great_northern_war_concluded
				NOT = {
					any_state = {
						is_core_of = FIN
						NOT = { is_owned_by = ROOT }
					}
				}
			}
		}
	}
	option = {
		name = TODRussia.5.a
		hidden_effect = {
			if = { limit = { country_exists = FIN } FIN = { remove_ideas = FIN_nordic_unification_offensive } }
			if = { limit = { country_exists = SWE } SWE = { remove_ideas = SWE_nordic_unification_offensive } }
		}
		if = {
			limit = {
				NOT = { has_global_flag = TOD_great_northern_war_concluded }
				country_exists = FIN
				NOT = { has_war_with = FIN }
			}
			declare_war_on = { target = FIN type = annex_everything }
			if = {
				limit = { country_exists = NOR NOT = { has_war_with = NOR } }
				declare_war_on = { target = NOR type = annex_everything }
			}
			if = {
				limit = { country_exists = SWE NOT = { has_war_with = SWE } }
				declare_war_on = { target = SWE type = puppet_wargoal_focus }
			}
		}
		else_if = {
			limit = {
				has_global_flag = TOD_great_northern_war_concluded
				any_state = {
					is_core_of = FIN
					is_owned_by = FIN
				}
				NOT = { has_war_with = FIN }
			}
			set_country_flag = TOD_RUS_northern_finished_winner_FIN
			declare_war_on = { target = FIN type = puppet_wargoal_focus }
		}
		else_if = {
			limit = {
				has_global_flag = TOD_great_northern_war_concluded
				any_state = {
					is_core_of = FIN
					is_owned_by = SWE
				}
				NOT = { has_war_with = SWE }
			}
			set_country_flag = TOD_RUS_northern_finished_winner_SWE
			declare_war_on = { target = SWE type = puppet_wargoal_focus }
		}
		add_war_support = 0.10
		set_country_flag = TOD_RUS_winter_war_started
	}
}

# 1939 年春：按十二种意识形态的实际支持率确认俄罗斯或苏俄路线。
country_event = {
	id = TODRussia.6
	title = TODRussia.6.t
	desc = TODRussia.6.d
	picture = GFX_report_event_chinese_soldiers_fighting
	fire_only_once = yes
	trigger = {
		tag = RUS
		date > 1939.2.28
		has_country_flag = TOD_RUS_winter_war_started
		has_country_flag = TOD_RUS_northern_settlement_completed
	}
	option = {
		name = TODRussia.6.a
		# 先锋社会主义必须是全部十二种意识形态中支持率最高的一种，才进入苏俄路线。
		set_temp_variable = { RUS_vanguard_support = party_popularity@vanguard_socialism }
		set_temp_variable = { RUS_national_corporatism_support = party_popularity@national_corporatism }
		set_temp_variable = { RUS_sacred_revivalism_support = party_popularity@sacred_revivalism }
		set_temp_variable = { RUS_leader_despotism_support = party_popularity@leader_despotism }
		set_temp_variable = { RUS_authoritarian_oligarchy_support = party_popularity@authoritarian_oligarchy }
		set_temp_variable = { RUS_conservative_democracy_support = party_popularity@conservative_democracy }
		set_temp_variable = { RUS_liberal_democracy_support = party_popularity@liberal_democracy }
		set_temp_variable = { RUS_consociationalism_support = party_popularity@consociationalism }
		set_temp_variable = { RUS_social_democrat_support = party_popularity@social_democrat }
		set_temp_variable = { RUS_agroindustrial_socialism_support = party_popularity@agroindustrial_socialism }
		set_temp_variable = { RUS_moderate_socialism_support = party_popularity@moderate_socialism }
		set_temp_variable = { RUS_council_socialism_support = party_popularity@council_socialism }
		if = {
			limit = {
				check_variable = { var = RUS_vanguard_support value = RUS_national_corporatism_support compare = greater_than }
				check_variable = { var = RUS_vanguard_support value = RUS_sacred_revivalism_support compare = greater_than }
				check_variable = { var = RUS_vanguard_support value = RUS_leader_despotism_support compare = greater_than }
				check_variable = { var = RUS_vanguard_support value = RUS_authoritarian_oligarchy_support compare = greater_than }
				check_variable = { var = RUS_vanguard_support value = RUS_conservative_democracy_support compare = greater_than }
				check_variable = { var = RUS_vanguard_support value = RUS_liberal_democracy_support compare = greater_than }
				check_variable = { var = RUS_vanguard_support value = RUS_consociationalism_support compare = greater_than }
				check_variable = { var = RUS_vanguard_support value = RUS_social_democrat_support compare = greater_than }
				check_variable = { var = RUS_vanguard_support value = RUS_agroindustrial_socialism_support compare = greater_than }
				check_variable = { var = RUS_vanguard_support value = RUS_moderate_socialism_support compare = greater_than }
				check_variable = { var = RUS_vanguard_support value = RUS_council_socialism_support compare = greater_than }
			}
			set_country_flag = TOD_RUS_soviet_route
			set_politics = {
				ruling_party = vanguard_socialism
				elections_allowed = no
			}
		}
		else = {
			set_country_flag = TOD_RUS_russian_route
		}
		set_country_flag = TOD_RUS_route_confirmed
	}
}

# 第一次北欧战争陷入僵持一年后，按苏芬战争式有限目标结算，阻止俄罗斯与北欧无限鏖战。
country_event = {
	id = TODRussia.13
	title = TODRussia.13.t
	desc = TODRussia.13.d
	picture = GFX_report_event_chinese_soldiers_fighting
	fire_only_once = yes
	mean_time_to_happen = { days = 1 }
	trigger = {
		tag = RUS
		has_country_flag = { flag = TOD_RUS_winter_war_started days > 365 }
		NOT = { has_country_flag = TOD_RUS_northern_settlement_completed }
		OR = {
			has_war_with = FIN
			has_war_with = SWE
			has_war_with = NOR
		}
	}
	option = {
		name = TODRussia.13.a
		# 俄罗斯攻占赫尔辛基：割让列宁格勒周边的芬兰卡累利阿。
		if = {
			limit = { 111 = { is_controlled_by = RUS } }
			RUS = { transfer_state = 146 }
146 = { set_state_controller_to = RUS }
		}
		# 北欧攻占圣彼得堡：以当前控制者获得完整科拉半岛。
		else_if = {
			limit = {
				195 = {
					controller = {
						OR = {
							tag = FIN
							tag = SWE
							tag = NOR
						}
					}
				}
			}
			195 = { controller = { save_event_target_as = TOD_RUS_nordic_winter_war_winner } }
			event_target:TOD_RUS_nordic_winter_war_winner = {
				transfer_state = 213
				transfer_state = 216
			}
		}
		# 其他情况视作无胜者，回到战前边界。
		if = { limit = { has_war_with = FIN } white_peace = FIN }
		if = { limit = { has_war_with = SWE } white_peace = SWE }
		if = { limit = { has_war_with = NOR } white_peace = NOR }
		set_country_flag = TOD_RUS_northern_settlement_completed
	}
}
